import requests
from bs4 import BeautifulSoup

risposta = requests.get("http://web-13.challs.olicyber.it/")
tag = BeautifulSoup(risposta.text, 'html.parser')

flag = ""

# per ogni tag <span> che abbia la classe "red"
for span in tag.find_all('span', class_='red'):
    flag = flag + span.text

print("flag{" + flag + "}")
