Inviare una richiesta con il verbo patch o put e stamparne l'header.

import requests

# URL della risorsa
url = "http://web-10.challs.olicyber.it/"

# Invia una richiesta PUT
risposta_put = requests.put(url)
print (risposta_put.headers)

risposta_patch = requests.patch(url)
print(risposta_patch.headers)
